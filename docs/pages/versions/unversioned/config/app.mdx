
import React, { useState } from 'react';
import {
  View,
  Text,
  Pressable,
  StyleSheet,
  SafeAreaView,
  StatusBar,
} from 'react-native';

export default function App() {
  const [taps, setTaps] = useState(0);
  const [record, setRecord] = useState(0);

  const handleTap = () => {
    setTaps(previous => {
      const next = previous + 1;
      setRecord(old => Math.max(old, next));
      return next;
    });
  };

  const formatNumber = number =>
    number.toLocaleString('en-US');

  return (
    <SafeAreaView style={styles.container}>
      <StatusBar barStyle="light-content" />

      <View style={styles.header}>
        <View>
          <Text style={styles.logo}>TAP<Text style={styles.accent}>RANK</Text></Text>
          <Text style={styles.subtitle}>GLOBAL TAP COMPETITION</Text>
        </View>
        <View style={styles.online}>
          <View style={styles.dot} />
          <Text style={styles.onlineText}>ONLINE</Text>
        </View>
      </View>

      <View style={styles.main}>
        <Text style={styles.label}>YOUR TOTAL TAPS</Text>

        <Text style={styles.counter}>
          {formatNumber(taps)}
        </Text>

        <Text style={styles.description}>
          Every tap takes you higher.
        </Text>

        <Pressable
          onPress={handleTap}
          style={({ pressed }) => [
            styles.tapButton,
            pressed && styles.tapPressed,
          ]}
        >
          <Text style={styles.tapText}>TAP!</Text>
          <Text style={styles.tapSubtext}>+1 POINT</Text>
        </Pressable>

        <Text style={styles.instruction}>
          TOUCH THE BUTTON TO SCORE
        </Text>
      </View>

      <View style={styles.stats}>
        <View style={styles.statCard}>
          <Text style={styles.statLabel}>SESSION RECORD</Text>
          <Text style={styles.statValue}>
            {formatNumber(record)}
          </Text>
        </View>

        <View style={styles.statCard}>
          <Text style={styles.statLabel}>YOUR RANK</Text>
          <Text style={styles.statValue}>--</Text>
        </View>
      </View>

      <View style={styles.footer}>
        <Text style={styles.footerText}>
          TAP RANK © 2026
        </Text>
        <Text style={styles.footerText}>
          COMPETE. TAP. DOMINATE.
        </Text>
      </View>
    </SafeAreaView>
  );
}
